
### Association (Uses-A Relationship)

```java
// Loose relationship
public class Driver {
    private String name;
    
    public Driver(String name) {
        this.name = name;
    }
    
    public void drive(Car car) {
        System.out.println(name + " is driving " + car.brand);
    }
}
```

### Aggregation (Has-A Relationship - Weak)

```java
// Department HAS-A Employee (but employee can exist without department)
public class Employee {
    private String name;
    
    public Employee(String name) {
        this.name = name;
    }
}

public class Department {
    private String name;
    private List<Employee> employees; // Aggregation
    
    public Department(String name) {
        this.name = name;
        this.employees = new ArrayList<>();
    }
    
    public void addEmployee(Employee emp) {
        employees.add(emp);
    }
}

// Usage
Employee emp1 = new Employee("John");
Employee emp2 = new Employee("Jane");
Department dept = new Department("IT");
dept.addEmployee(emp1);
dept.addEmployee(emp2);
// Employees exist even if department is deleted
```

### Composition (Has-A Relationship - Strong)

```java
// House HAS-A Room (Room cannot exist without House)
public class Room {
    private String type;
    private double size;
    
    public Room(String type, double size) {
        this.type = type;
        this.size = size;
    }
}

public class House {
    private String address;
    private List<Room> rooms; // Composition
    
    public House(String address) {
        this.address = address;
        this.rooms = new ArrayList<>();
        // Rooms created WITH the house
        rooms.add(new Room("Bedroom", 20.5));
        rooms.add(new Room("Kitchen", 15.0));
        rooms.add(new Room("Living Room", 30.0));
    }
    
    public void displayRooms() {
        System.out.println("House at " + address + " has:");
        for (Room room : rooms) {
            System.out.println("- " + room.type + " (" + room.size + " sq ft)");
        }
    }
}

// Usage
House house = new House("123 Main St");
house.displayRooms();
// If house is destroyed, rooms are destroyed too
```
